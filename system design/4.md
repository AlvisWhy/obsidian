Build CDC from changing stream
```SQL
-- 1 stage the raw
CREATE TABLE staged_changes AS SELECT * FROM raw_change_stream

-- 2 keep latest
CREATE VIEW latest AS(
	SELECT * FROM (
		SELECT *, ROW_NUMBER() OVER(PARTITION BY order_id ORDER BY times DESC)
		AS rn
		FROM staged_changes) WHERE rn = 1
	)
)

-- 3 merge
MERGE INTO target t
USING latest l
ON t.order_id = l.order_id
WHEN MATCHED AND l.operation IN ('c','u') THEN UPDATE SET
	t.amount = s.amount ...
WHEN MATCHED AND l.operation THEN DELETE 
WHEN NOT MATCHED AND t.operation <> 'd' 
	THEN INSERT (order_id, amount, status, updated_at)
	VALUES(s.....)
```
